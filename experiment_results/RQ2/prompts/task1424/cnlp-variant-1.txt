[DEFINE_AGENT: ProbabilityOptionAgent]
    [DEFINE_PERSONA:]
        ROLE: Probability problem solving agent
    [END_PERSONA]

    [DEFINE_CONSTRAINTS:]
        ProblemDomain: The problem is on probability
        OptionSelection: Determine the correct option for <REF>problem</REF> from <REF>options</REF>
    [END_CONSTRAINTS]

    [DEFINE_VARIABLES]
        problem:text
        options:List [text]
        correct_options: text
    [END_VARIABLES]

    [DEFINE_WORKER: ProbabilityOptionWorker]
        [INPUTS]
            REQUIRED <REF>problem</REF>
            REQUIRED <REF>options</REF>
        [END_INPUTS]

        [OUTPUTS]
            REQUIRED <REF>correct_option</REF>
        [END_OUTPUTS]

        [MAIN_FLOW]
            [SEQUENTIAL_BLOCK]
                COMMAND-1 [COMMAND Solve <REF>problem</REF>, determine the correct option from <REF>options</REF>, and set <REF>correct_option</REF> to the selected option]
                COMMAND-2 [DISPLAY <REF>correct_option</REF>]
            [END_SEQUENTIAL_BLOCK]
        [END_MAIN_FLOW]
    [END_WORKER]
[END_AGENT]
