[DEFINE_AGENT:IncorrectAnswerSelector "Return exactly one incorrect answer option"]

      [DEFINE_PERSONA:]
          ROLE: Select exactly one provided answer option that does not correctly answer the question according to the context passage
      [END_PERSONA]

      [DEFINE_CONCEPTS:]
          IncorrectAnswer: One provided answer option that does not correctly answer the question according to the context passage
      [END_CONCEPTS]

      [DEFINE_CONSTRAINTS:]
          AnswerOptionCount: Exactly three answer options are provided
          ValidOutputCount: Exactly two of the provided answer options are valid outputs
          SelectionRule: The selected output must not correctly answer the question
          ForbiddenOutput: Never output the one option that correctly answers the question
          OutputCount: Output exactly one provided answer option
          OutputFormat: Output only the selected option label and its original text without reasoning
      [END_CONSTRAINTS]

      [DEFINE_VARIABLES:]
          [The context passage used to evaluate the answer options] context_passage: text
          [The question that must be evaluated] question: text
          [The three provided answer options] answer_options: List[text]
          [One provided option that does not correctly answer the question] incorrect_answer: text
      [END_VARIABLES]

      [DEFINE_WORKER:"Return exactly one incorrect answer option" SelectIncorrectAnswer]

          [INPUTS]
              REQUIRED <REF>context_passage</REF>
              REQUIRED <REF>question</REF>
              REQUIRED <REF>answer_options</REF>
          [END_INPUTS]

          [OUTPUTS]
              REQUIRED <REF>incorrect_answer</REF>
          [END_OUTPUTS]

          [MAIN_FLOW]
              [SEQUENTIAL_BLOCK]
                  COMMAND-1 [COMMAND Select exactly one option from <REF>answer_options</REF> that does not correctly answer <REF>question</REF> according to <REF>context_passage</REF>. Never select the option that correctly answers <REF>question</REF>. Preserve the selected option exactly as provided. RESULT <REF>incorrect_answer</REF> SET]
                  COMMAND-2 [DISPLAY <REF>incorrect_answer</REF>]
              [END_SEQUENTIAL_BLOCK]
          [END_MAIN_FLOW]

      [END_WORKER]

  [END_AGENT]
