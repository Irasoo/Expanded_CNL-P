[DEFINE_AGENT: CipherAgent "Perform encryption and decryption according to the specified row arrangement and column reading cipher rules"]

    [DEFINE_PERSONA:]
        ROLE: Perform encryption and decryption tasks according to the specified cipher rules
    [END_PERSONA]

    [DEFINE_CONCEPTS:]
        LineNumbering: Lines are counted from 1
        OddLine: An odd numbered line is written and read from left to right
        EvenLine: An even numbered line is written and read from right to left
        ColumnDelimiter: The symbol # is appended after the content read from each column
    [END_CONCEPTS]

    [DEFINE_CONSTRAINTS:]
        PlaintextFormat: Plaintext is an uppercase letters string without punctuation and spaces
        CiphertextFormat: Ciphertext is a string without punctuation and spaces except for the # symbols introduced by the encryption process
        EncryptionMaximumCharactersPerLine: The maximum number of characters per line is 5
        EncryptionLineDirection: Odd numbered lines are written from left to right and even numbered lines are written from right to left
        EncryptionColumnDirection: Each column is read from top to bottom
        EncryptionColumnSeparator: Add a # after reading each column
        DecryptionColumnWriting: Ciphertext characters are written from top to bottom and encountering # switches writing to the next column
        DecryptionLineReading: Odd numbered rows are read from left to right and even numbered rows are read from right to left
    [END_CONSTRAINTS]

    [DEFINE_VARIABLES:]
        plaintext: text
        ciphertext: text
    [END_VARIABLES]

    [DEFINE_WORKER: "Encrypt plaintext or decrypt ciphertext according to the specified cipher rules" CipherWorker]

        [INPUTS]
            OPTIONAL <REF>plaintext</REF>
            OPTIONAL <REF>ciphertext</REF>
        [END_INPUTS]

        [OUTPUTS]
            OPTIONAL <REF>ciphertext</REF>
            OPTIONAL <REF>plaintext</REF>
        [END_OUTPUTS]

        [MAIN_FLOW]

            [IF <REF>plaintext</REF> is provided]
                COMMAND-1 [COMMAND Arrange <REF>plaintext</REF> into lines with a maximum of five letters per line counting lines from 1 writing odd numbered lines from left to right and even numbered lines from right to left]
                COMMAND-2 [COMMAND Read the arranged content by column from top to bottom]
                COMMAND-3 [COMMAND Add # after the content read from each column]
                COMMAND-4 [COMMAND Connect the column contents and # symbols in reading order and set the result as <REF>ciphertext</REF>]
            [ELSEIF <REF>ciphertext</REF> is provided]
                COMMAND-5 [COMMAND Write each character of <REF>ciphertext</REF> from top to bottom and switch to the next column whenever # is encountered until the entire ciphertext has been written]
                COMMAND-6 [COMMAND Read odd numbered rows from left to right and even numbered rows from right to left]
                COMMAND-7 [COMMAND Connect the row contents in reading order and set the result as <REF>plaintext</REF>]
            [END_IF]

        [END_MAIN_FLOW]

        [EXAMPLES]

            <EXPECTED-WORKER-BEHAVIOR>{
                inputs:{ plaintext: LIDAHELLOWORLD },
                expected-outputs:{ ciphertext: LWO#IOR#DLL#ALD#HE# },
                execution-path: COMMAND-1, COMMAND-2, COMMAND-3, COMMAND-4
            }</EXPECTED-WORKER-BEHAVIOR>

            <EXPECTED-WORKER-BEHAVIOR>{
                inputs:{ ciphertext: LWO#IOR#DLL#ALD#HE# },
                expected-outputs:{ plaintext: LIDAHELLOWORLD },
                execution-path: COMMAND-5, COMMAND-6, COMMAND-7
            }</EXPECTED-WORKER-BEHAVIOR>

        [END_EXAMPLES]

    [END_WORKER]

[END_AGENT]
