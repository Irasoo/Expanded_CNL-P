[DEFINE_AGENT: CipherAgent "Perform encryption and decryption according to the specified row arrangement and column reading cipher rules"]

    [DEFINE_PERSONA:]
        ROLE:You are an intelligent assistant that specializes in encryption and decryption tasks.
    [END_PERSONA]

    [DEFINE_CONSTRAINTS:]
        PlaintextFormat: Plaintext is an uppercase letters string without punctuation and spaces
        CiphertextFormat: Ciphertext is a string without punctuation and spaces except for the # symbols introduced by the encryption process
        OutputCompliance: All encryption and decryption outputs must adhere to the specified cipher rules and required output format
    [END_CONSTRAINTS]

		[DEFINE_VARIABLES:]
		    plaintext: text
		    ciphertext: text
		    arranged_rows: List[text]
		    row_segments: List[text]
		    column_segments: List[text]
		[END_VARIABLES]

    [DEFINE_WORKER: "Encrypt plaintext or decrypt ciphertext according to the specified cipher rules" CipherWorker]

        [INPUTS]
            OPTIONAL <REF>plaintext</REF>
            OPTIONAL <REF>ciphertext</REF>
        [END_INPUTS]

        [OUTPUTS]
            OPTIONAL <REF>ciphertext</REF>
            OPTIONAL <REF>plaintext</REF>
        [END_OUTPUTS]

        [MAIN_FLOW]
           [IF <REF>plaintext</REF> is provided]
                COMMAND-1 [COMMAND Arrange <REF>plaintext</REF> into lines with a maximum of five letters per line counting lines from 1 writing odd numbered lines from left to right and even numbered lines from right to left RESULT <REF>arranged_rows</REF> SET]
                COMMAND-2 [COMMAND Read <REF>arranged_rows</REF> column by column from left to right reading each column from top to bottom and appending # after the content read from each column RESULT <REF>column_segments</REF> SET]
                COMMAND-3 [COMMAND Connect the elements of <REF>column_segments</REF> in column order RESULT <REF>ciphertext</REF> SET]
                COMMAND-4 [DISPLAY <REF>ciphertext</REF>]
            [ELSEIF <REF>ciphertext</REF> is provided]
                COMMAND-5 [COMMAND Write each non # character of <REF>ciphertext</REF> from top to bottom in the current column and switch to the next column whenever # is encountered until the entire ciphertext has been processed RESULT <REF>arranged_rows</REF> SET]
                COMMAND-6 [COMMAND Read <REF>arranged_rows</REF> in row order reading odd numbered rows from left to right and even numbered rows from right to left RESULT <REF>row_segments</REF> SET]
                COMMAND-7 [COMMAND Connect the elements of <REF>row_segments</REF> in row order RESULT <REF>plaintext</REF> SET]
                COMMAND-8 [DISPLAY <REF>plaintext</REF>]
            [END_IF]
        [END_MAIN_FLOW]

        [EXAMPLES]

            <EXPECTED-WORKER-BEHAVIOR>{
                inputs:{ plaintext: LIDAHELLOWORLD },
                expected-outputs:{ ciphertext: LWO#IOR#DLL#ALD#HE# },
                execution-path: COMMAND-1, COMMAND-2, COMMAND-3, COMMAND-4
            }</EXPECTED-WORKER-BEHAVIOR>

            <EXPECTED-WORKER-BEHAVIOR>{
                inputs:{ ciphertext: LWO#IOR#DLL#ALD#HE# },
                expected-outputs:{ plaintext: LIDAHELLOWORLD },
                execution-path: COMMAND-5, COMMAND-6, COMMAND-7, COMMAND-8
            }</EXPECTED-WORKER-BEHAVIOR>

        [END_EXAMPLES]

    [END_WORKER]

[END_AGENT]
