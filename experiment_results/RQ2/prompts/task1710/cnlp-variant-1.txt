    [DEFINE_WORKER: """Encrypt plaintext or alternatively decrypt ciphertext using the specified reels""" ProcessReelCipher]

        [INPUTS]
            OPTIONAL <REF>plaintext</REF>
            OPTIONAL <REF>ciphertext</REF>
        [END_INPUTS]

        [OUTPUTS]
            OPTIONAL <REF>ciphertext</REF>
            OPTIONAL <REF>plaintext</REF>
        [END_OUTPUTS]

        [MAIN_FLOW]
            [SEQUENTIAL_BLOCK]
                COMMAND-1 [COMMAND Set <REF>ciphertext</REF> to empty text and initially select Reel1]
            [END_SEQUENTIAL_BLOCK]

            [FOR each character p in <REF>plaintext</REF> in order]
                COMMAND-2 [COMMAND Find p on the current Reel and set <REF>ciphertext</REF> to <REF>ciphertext</REF> followed by the next character on the current Reel or the first character on the current Reel if p is at the end]
                COMMAND-3 [COMMAND Move to the next Reel or circle back to Reel1 if the current Reel is Reel25]
            [END_FOR]

            [SEQUENTIAL_BLOCK]
                COMMAND-4 [DISPLAY <REF>ciphertext</REF>]
            [END_SEQUENTIAL_BLOCK]
        [END_MAIN_FLOW]


        [ALTERNATIVE_FLOW: """Decrypt ciphertext instead of encrypting plaintext""" DecryptCiphertext]
            [SEQUENTIAL_BLOCK]
                COMMAND-5 [COMMAND Set <REF>plaintext</REF> to empty text and initially select Reel1]
            [END_SEQUENTIAL_BLOCK]

            [FOR each character c in <REF>ciphertext</REF> in order]
                COMMAND-6 [COMMAND Find c on the current Reel and set <REF>plaintext</REF> to <REF>plaintext</REF> followed by the previous character on the current Reel or the last character on the current Reel if c is at the beginning]
                COMMAND-7 [COMMAND Move to the next Reel or circle back to Reel1 if the current Reel is Reel25]
            [END_FOR]

            [SEQUENTIAL_BLOCK]
                COMMAND-8 [DISPLAY <REF>plaintext</REF>]
            [END_SEQUENTIAL_BLOCK]
        [END_ALTERNATIVE_FLOW]

    [END_WORKER]
